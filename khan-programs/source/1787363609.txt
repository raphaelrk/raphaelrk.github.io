 * Feel free to use this for your own programs, but please

 * include this link:

 * http://www.khanacademy.org/cs/drawbutton/1082731226

 * To use it, copy everything except the draw loop, and paste

 * it in your program.

 * This function takes 5 parameters: text, x-position,

 * y-position, width (or length), and color.

 * If the mouse is over the button, over_button is set to

 * be true; otherwise, false. Immediately after the function

 * is called, you can use over_button for other purposes,

 * like a button_clicked function.

*/



var over_button; //After the function is called, this variable will be either true or false.



var drawButton = function(button_text, x, y, length, color) {

    

    var button_height = 22;

    

    over_button = false;

    var outline = color;

    var greyV = 180; // Lightness of the button.

    var greyVB = 220; // Lightness of the button when the mouse is over it

    var Val = greyV;   

    

    x -= length/2;

            

    //Button's Side

    strokeWeight(1);

    stroke(0, 0, 0, 100);

    fill(outline);

    rect(x-2, y+2, length, button_height, 3);

         

    // If hovered over

    if (mouseX >= x && mouseX <= x + length && mouseY >= y && mouseY <= y + button_height) {

        Val = greyVB;

        over_button = true;

        if (mouseIsPressed) {

                x -=2;

                y += 2;

        }

    }

    fill(random(0,255), random(0,255), random(0,255));

    textSize(20);

    text("The Filter(); Function!",100,100);

    

    textSize(13);

    text("Try to figure out the code for anti-blurry without looking in the code!",12,123);

    

    

    // Button Face

    strokeWeight(2);

    stroke (outline);

    fill(Val, Val, Val);

    rect(x, y, length, button_height, 2);

    

    // Text

    textAlign(CENTER,CENTER);

    fill (0, 102, 19);

    textSize(12);

    text (button_text, x + length/2, y + button_height/2);

    textAlign(LEFT, 0);

};

var b;

r = round(random(0,1)) * 255;

g = round(random(0,1)) * 255;

b = round(random(0,1)) * 255;

if (r === g && g === b) {

    var rnd = round(random(1, 3));

    switch(rnd) {

        case 1:

            r = 0;

            g = 255;

            break;

        case 2:

            g = 0;

            b = 255;

            break;

        case 3:

            b = 0;

            r = 255;

    }

}
