var keyPressed = function() {
    if (keyCode === CONTROL) {
        CTRL = true;
    }
    if (typing) {
        switch(keyCode) {
        case LEFT:
            if (disp) {
                typePos = start;
            } else if (typePos > 0) {
                typePos --;
            }
            caret = true;
            exitHighlight();
            break;
        case RIGHT:
            if (disp) {
                typePos = end;
            } else if(typePos < entry.length) {
                typePos ++;
            }
            caret = true;
            exitHighlight();
            break;
        case DOWN:
            entry = "";
            caret = true;
            exitHighlight();
            break;
        case BACKSPACE: case DELETE:
            if (disp) {
                entry = join(concat(subset(split(entry, ""), 0, start), subset(split(entry, ""), end, entry.length - end)), "");
                typePos = start;
            } else if(typePos > 0) {
                entry = join(concat(subset(split(entry, ""), 0, typePos - 1), subset(split(entry, ""), typePos, entry.length - typePos)), "");
                typePos --;
            }
            caret = true;
            exitHighlight();
            break;
            case ENTER:
                typing = false;
        }
        if (CTRL) {
            switch (key.toString().toLowerCase()) {
                case "a":
                    disp = true;
                    a = 0;
                    b = entry.length;
                    break;
                case "x":
                    if (disp) {
                        entry = join(concat(subset(split(entry, ""), 0, start), subset(split(entry, ""), end, entry.length - end)), "");
                    typePos = start;
                        exitHighlight();
                    }
                    break;
            }
        } else if (allowedKeys.indexOf(key.toString()) !== -1) {
            if (disp) {
                entry = join(concat(append(subset(split(entry, ""), 0, start), key.toString()), subset(split(entry, ""), end, entry.length - end)), "");
                typePos = start + 1;
            } else {
                entry = join(splice(split(entry, ""), key.toString(), typePos), "");
                typePos ++;
            }
            caret = true;
            exitHighlight();
        }
    }
};

var updatePlotHistory = function() {
    var p = [];
        for (var b = 0; b < pixel.length; b ++) {
            p[b] = [];
            if (pixel[b] !== undefined) {
                for (var a = 0; a < pixel[b].length; a ++) {
                    p[b][a] = pixel[b][a];
                }
            }
        }
    var plotObject = {
        width: grid.width,
        height: grid.height,
        pixel: p
    };
    if (plotHistory[plotHistory.length - 1] !== plotObject) {
        if (plotHistory.length >= 20) {
            plotHistory = append(subset(plotHistory, 1, plotHistory.length - 1), plotObject);
        } else {
            plotHistory.push(plotObject);
        }
    }
};

var insideEllipse = function(a, b, x1, y1, x2, y2) {
    if (4 * sq(a - (x1 + x2)/2)/sq(x2 - x1) + 4 * sq(b - (y1 + y2)/2)/sq(y2 - y1) <= 1 + 2/(x2 - x1)) {
        return true;
    } else {
        return false;
    }
};

var onEllipse = function(a, b, x1, y1, x2, y2) {
    if (4 * sq(a - (x1 + x2)/2)/sq(x2 - x1) + 4 * sq(b - (y1 + y2)/2)/sq(y2 - y1) >= 1 - 2/(x2 - x1) && 4 * sq(a - (x1 + x2)/2)/sq(x2 - x1) + 4 * sq(b - (y1 + y2)/2)/sq(y2 - y1) <= 1 + 2/(x2 - x1)) {
        return true;
    } else {
        return false;
    }
};

var inShape = function(a, b, x1, y1, x2, y2) {
    if (drawMode === "rectangle") {
        if (shapeFill) {
            return true;
        } else if (a === x1 || a === x2 || b === y1 || b === y2) {
            return true;
        } else {
            return false;
        }
    } else if (drawMode === "ellipse") {
        if (shapeFill) {
            if (insideEllipse(a, b, x1, y1, x2, y2)) {
                return true;
            } else {
                return false;
            }
        } else if (onEllipse(a, b, x1, y1, x2, y2)) {
            return true;
        } else {
            return false;
        }
    }
};

var dashedLineH = function(y, xA, xB) {
    var x1 = min(xA, xB);
    var x2 = max(xA, xB);
    var m = 6;
    var n = round((x2 - x1)/m);
    n += (n % 2) - 1;
    var w = (x2 - x1)/max(n, 1);
    var t = 0;
    for (var i = x1; i < x2 - w; i += w) {
        if (t % 2 === 0) {
            line(i, y, i + w, y);
        }
        t ++;
    }
};

var dashedLineV = function(x, yA, yB) {
    var y1 = min(yA, yB);
    var y2 = max(yA, yB);
    var m = 6;
    var n = round((y2 - y1)/m);
    n += (n % 2) - 1;
    var w = (y2 - y1)/max(n, 1);
    var t = 0;
    for (var i = y1; i < y2 - w; i += w) {
        if (t % 2 === 0) {
            line(x, i, x, i + w);
        }
        t ++;
    }
};

    var y = 130;
    var w = 24;
    var h = 14;

    background(255);

    fill(0);
    if (f > 30) {

    } else if (f < 130) {
        for (var b = 0; b < h; b ++) {
            for (var a = 0; a < w; a ++) {
                if (pA[b][a] <= f) {
                    rect(x + a * s, y + b * s, s, s);

        }

    }

                if (pA[b][a] <= f - 100) {
                    rect(x + a * s, y + b * s, s, s);

            }
        }

var pcursorState;
var updateCursor = function() {
    switch (cursorState) {
        case true: cursor(); break;
        case false: noCursor(); break;
        default: this.cursor(cursorState);
    }
};

var onScreen = false;
var updateMouse = function() {
    pmouseIsPressed = mouseIsPressed;
    if (onScreen) {
        updateCursor();
    }
    if (mouseX !== pmouseX || mouseY !== pmouseY) {
        onScreen = true;
    }
};
var updateKeys = function() {
    if (!keyIsPressed) {
        endPress = true;
    }
};

var dispMode = function() {
    switch (page) {
        case "intro": intro(); break;
        case "edit": editMode(); break;
        case "color": editColor(); break;
        case "preview": previewMode(); break;
        default: goToPage("edit");
    }
};

var update = function() {
    messageSet = "";
    if (clearPage) {
        background(BACKGROUND);
        clearPage = false;
        rEP = true;
    }
    if (ppage === 1) {
        ppage = page;
    }
    if (ppage !== page) {
        ppage = 1;
    }
};

var draw = function() {
    setCursor(true);
    update();
    dispMode();
    updateMouse();
    updateKeys();
};
