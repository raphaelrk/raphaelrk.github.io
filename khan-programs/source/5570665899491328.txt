var print_center = function(msg) {
  var numspaces=(MAXLINELEN-msg.length)/2;
  
  for(var i=0;i<numspaces;i++) {
    print(" ");
  }
  print(msg + '\n');
};

var print_title = function() {
    print("\n\n");
    print_center("*** ELIZA ***");
    print_center("Original code by Weizenbaum, 1966");
    print_center("To stop Eliza, type 'bye'");
    print("\n\n");
    print("HI!  I'M ELIZA.  WHAT'S YOUR PROBLEM?\n");
};

var k,baseLength; 
var whichReply = [];
var lastinput = "";
var reply = "";
var baseResponse, token;
var separator=" ";
var inputstr = "";

var main = function()
{
    // use the first reply for each keyword match the first time you see that keyword
    for (var x=0; x<NUMKEYWORDS; x++) {
	    whichReply[x] = 0;
    }

    // print a nice centered title screen
    print_title();

    lastinput[0]='\0';
    
    while (true) { 
        //readline(inputstr);

	    // check for termination 
	    if (inputstr === "BYE")
	    {
		    break; 
	    }
	    
        // check for repeated entries 
	    if (lastinput === inputstr) 
	    {
		    reply = "PLEASE DON'T REPEAT YOURSELF!";
		    //continue;
		    break;
	    }
	    
        lastinput = inputstr;
        
        // see if any of the keywords is contained in the input 
        // if not, we use the last element of keywords as our default responses 
        var loc;
	    reply = "";
        for(k=0;k<NUMKEYWORDS-1;k++)
        {
            loc=inputstr.indexOf(keywords[k]); 
            if(loc !== -1)
            {
				break;
            }
        }
        
        // Build Eliza's response 
        // start with Eliza's canned response, based on the keyword match
		baseResponse = responses[k][whichReply[k]];
		baseLength = baseResponse.length;

		if(baseResponse[baseLength-1] !== '*')
		{
		    // if we have a baseResponse without an asterix, just use it as-is
            reply += baseResponse;
		}
		else
		{
            // if we do have an asterix, fill in the remaining with the user input
		    // use all but the last character of the base response
		    reply += baseResponse.substring(0, baseResponse.length-2);

		    // now add in the rest of the user's input, starting at <location>
		    // but skip over the keyword itself
		    loc+=keywords[k].length;
		    // take them one word at a time, so that we can substitute pronouns
		    var tokens = inputstr.substring(loc).split(" ");
		    for(var i=0; i < tokens.length; i++)
		    {
		        for(var s=0;s<NUMSWAPS;s++)
                {   
                    if(SWAPS[s][0] === tokens[i])
                    {
                        tokens[i]=SWAPS[s][1];
                        break;
                    }
                }   
                reply += " ";
                reply += tokens[i];
		    }
		    
            reply += "?";
		}
        // println(reply);
	     
	    // next time, use the next appropriate reply for that keyword
	    whichReply[k]++;
	    if (whichReply[k] >= ResponsesPerKeyword[k])
	    {
		    whichReply[k] = 0;
	    }
	    
	    break;
    } 

    reply += "GOODBYE!  THANKS FOR VISITING WITH ME...";
};

var draw = function() {
    textAlign(CENTER);
    text(inputstr, 200, 200);
    text(reply, 200, 200);
};
