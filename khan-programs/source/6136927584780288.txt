/** 
 * sets/resets scene to be the sceneObject
 * 
 * resets coloring, text, angle, framerate, and transformation changes
 * doesn't reset changes to random or noise
 * */
var setScene = function(sceneObject) {
    // reset colors
    colorMode(RGB, 255);
    background(255, 255, 255);
    fill(255, 255, 255);
    stroke(0, 0, 0);
    strokeWeight(1);
    
    // reset text
    textFont(createFont("sans-serif"), 12);
    textAlign(LEFT, BOTTOM);
    
    // reset angle mode
    angleMode = "degrees";
    
    // reset framerate and transformation
    frameRate(30);
    resetMatrix();
    
    // reset to loop mode
    loop();
    
    // set the scene to be one of sceneObject
    scene = new sceneObject();
};
