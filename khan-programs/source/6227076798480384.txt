/**
 * 10 PRINT
 * By Raphael Rouvinov-Kats
 * Based on the famous commodore 64 one-liner "10 PRINT CHR$(205.5+RND(1)); : GOTO 10"
 * */

background(141, 187, 240);

stroke(255, 255, 255);
strokeWeight(5);

var s = 20;

for(var y = 0; y < 400; y += s) {
    for(var x = 0; x < 400; x+= s) {
        if(random(0, 1) < 0.5) {
            line(x, y, x + s, y + s);
        } else {
            line(x + s, y, x, y + s);
        }
    }
}

frameRate(1);
