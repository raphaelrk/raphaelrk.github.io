var getAngle = function(x1, y1, x2, y2) {
        var xdiff = abs(x2 - x1); // adj
        var ydiff = abs(y2 - y1); // opp
        
        var angle;
        if(x1 > x2) {
            if(y2 < y1) {
                angle = -atan(ydiff/xdiff) + 90;
            } else {
                angle = atan(ydiff/xdiff) + 90;
            }
        } else {
            if(y1 < y1) {
                angle = atan(ydiff/xdiff) - 90;
            } else {
                angle = -atan(ydiff/xdiff) - 90;
            }
        }
        
        return angle;
    };

var dart = function(x, y, dx, dy) {
    this.x = x;
    this.y = y;
    this.dx = dx;
    this.dy = dy;
    var angle = atan(dy/dx);
    
    this.update = function() {
        x += dx;
        y += dy;
        dx = max(1, dx - 0.1);
        dy = dy + 1;
        angle = atan(dy/dx);
        angle = getAngle(x, y, x + dx, y + dy);
    };
    
    this.draw = function() {
        pushMatrix();
        translate(x, y);
        rotate(angle);
        ellipse(0, 0, 4, 10);
        popMatrix();
    };
    
    this.offscreen = function() {
        return false;
    };
};

                angle = atan(ydiff/xdiff) - 90;
            } else {
                angle = -atan(ydiff/xdiff) - 90;

var bloon = function(x, y) {
    this.x = x;
    this.y = y;
    
    this.draw = function() {
        fill(191, 70, 70);
        ellipse(x, y, 20, 20);
    };
    
    this.getX = function() {return x;};
    this.getY = function() {return y;};
};

var level = function(arr) {
    var bloons = [];
    for(var i = 0; i < arr.length; i++) {
        bloons.push(new bloon(arr[i][0], arr[i][1]));
    }
    
    this.draw = function() {
        for(var b = 0; b < bloons.length; b++) {
            bloons[b].draw();
        }
    };
    
    this.free = function() {
        for(var i = 0; i < bloons.length; i++) {
            bloons.pop();
        }
    };
};
