/**
 * This changes the background
 * The r is for red
 * The g is for green
 * The b is for blue
 * */
var drawBG = function(r, g, b) {
    
    /**
     * These are the speed of the change in color for
     * the background.
     * rcspeed is how quickly the red changes
     * gcspeed is how quickly the green changes
     * bcspeed is how quickly the blue changes
     * */
    var rcspeed = 0.5;
    var gcspeed = 0.1;
    var bcspeed = 0.1;
    
    /**
     * These are the magnitude of the change in color for
     * the background.
     * rmagnitude is how much the red changes
     * gmagnitude is how much the green changes
     * bmagnitude is how much the blue changes
     * */
    var rmagnitude = 50;
    var gmagnitude = 50;
    var bmagnitude = 50;
    
    /**
     * These values are here so that what would normally be
     * a bland background really brightens up with some 
     * change (that's what the c is for)
     * */
    var rc = -rmagnitude*sin(millis()*rcspeed);
    var gc = gmagnitude*cos(millis()*gcspeed);
    var bc = bmagnitude*sin(millis()*bcspeed);
    
    background(r + rc, g + gc, b + bc);
};

/**

    textFont(one_font, one_size); // the font!

                mouseY + one_mousey_offset);
};

/**
 * This is the central hub of this program
 * This function is repeatedly called following 
 * initialization.
 * If you want to know how the program works, you look here
 * */
var draw = function() {
    drawBG(bg_r, bg_g, bg_b);
    drawOne();
};
