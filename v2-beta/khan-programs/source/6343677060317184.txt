var col1 = new color(255, 255, 255);
var col2 = color(56, 56, 56);
var d = 40;

var drawCross = function(c) {
    fill(c);
    stroke(c);
    
    var t = d/3; // size of one block
    rect(-t/2, -t/2, t, t); // center
    rect(-t/2, -t/2 - t, t, t); // top
    rect(-t/2, -t/2 + t, t, t); // bottom
    rect(-t/2 - t, -t/2, t, t); // left
    rect(-t/2 + t, -t/2, t, t); // right
};

var draw = function() {
    background(col1);
    translate(d/2, d/2);
    for(var y = 0; y < 400 + d; y += d) {
        for(var x = 0; x < 400 + d; x += d) {
            pushMatrix();
                translate(x / 2.0, y / 2.0);
                rotate(frameCount);
                drawCross(col2);
            popMatrix();
        }
    }
};
